\begin{frame}{IDQNN 的输入-输出分布结构}
  \begin{block}{本文问题设定（明确）}
    本文考虑的经典比特串生成任务由 \textbf{IDQNN} 完成，但目标是\textbf{特殊结构分布}，而不是一般任意分布。
  \end{block}

  \begin{itemize}
    \item 当输入比特串 $x=\mathbf{1}$ 时，生成的 $y$ 为均匀分布；
    \item 当输入比特串 $x=\mathbf{0}$ 时，生成的 $y$ 的前 $((n_1-1)\times m)$ 个比特的边缘分布为均匀分布。
  \end{itemize}

  \begin{alertblock}{结论}
    该模型与训练方案\textbf{不能直接覆盖任意数据集}（例如手写数字识别数据集）上的一般分布学习任务。
  \end{alertblock}
\end{frame}

\section{数据与训练设定}

\begin{frame}{数据集构造与训练目标}
  数据并非任意采集，而是由“理想参数 + 指定输入分布”生成：
  \begin{itemize}
    \item 预先固定一组理想参数 $\theta_{\mathrm{real}}$；
    \item 给定输入比特串分布 $D(x)$（文中实验设定）：
    \begin{itemize}
      \item 以 $\frac{1}{3}$ 概率，取 $x=0^n$；
      \item 以 $\frac{1}{3}$ 概率，$x$ 的每一比特独立采样，且 $P(x_j=0)=0.6,\;P(x_j=1)=0.4$；
      \item 以 $\frac{1}{3}$ 概率，$x$ 的每一比特独立采样，且 $P(x_j=0)=0.2,\;P(x_j=1)=0.8$。
    \end{itemize}
    \item 重复采样 $N$ 次得到
    \[
      \text{dataset}=\{(x_i,y_i)\}_{i=1}^{N}.
    \]
  \end{itemize}
   训练任务为：
  \begin{itemize}
    \item \textbf{网络结构已知}（IDQNN 结构）的情况下；
    \item 学习参数 $\theta$，使其逼近目标参数 $\theta_{\mathrm{real}}$。
  \end{itemize}

\end{frame}

\begin{frame}{本文学习算法：样本筛选与参数估计}
  \small
  	\textbf{1) 样本筛选（以第 $(i,j)$ 个参数为例）}
  \begin{itemize}
    \item 从 $\text{dataset}=\{(x_i,y_i)\}_{i=1}^N$ 中筛选满足条件的子集：
    \[
      x_{i,j}=0,\quad x_{\mathcal N(i,j)}=1,
    \]
    其中 $\mathcal N(i,j)$ 表示第 $(i,j)$ 位的邻域比特；
    \item 记筛选后样本数为 $N_{\mathrm{sp}}$，对应输出位记作
    \[
      y^{(i,j)}_{(t)}\in\{0,1\},\quad t=1,\dots,N_{\mathrm{sp}}.
    \]
  \end{itemize}

  \vspace{0.2em}
  	\textbf{2) 估计参数}
  \[
    \hat\theta_{i,j}=\frac{1}{2}\arccos\!\left(1-\frac{2}{N_{\mathrm{sp}}}\sum_{t=1}^{N_{\mathrm{sp}}}y^{(i,j)}_{(t)}\right).
  \]

  \vspace{0.2em}
  	\textbf{3) 估计误差}
  \[
    \mathrm{Std}(\hat\theta_j)\approx \frac{1}{2\sqrt{N_{\mathrm{sp}}}},
    \qquad
    |\hat\theta_j-\theta_j|=\mathcal O\!\left(\frac{1}{\sqrt{N_{\mathrm{sp}}}}\right).
  \]

\end{frame}


\begin{frame}{代码现状：模拟采样}
  现有采样代码（DQNN \& IDQNN）基于 \texttt{cirq} simulator：
  \begin{itemize}
    \item 本质是对量子算法的经典仿真；
    \item 时间与空间开销随比特数呈指数增长。
  \end{itemize}

  即使使用 tensor network 等技巧，也难以对“所有输入 $x$”普遍降低模拟复杂度。
  $$\mathcal{L}_{\text{NLL}}(\theta) = -\sum_{i=1}^N \log p_\theta(y_i \mid x_i)$$
  $$\theta_{n+1} = \theta_n - \eta \nabla_\theta \mathcal{L}_{\text{NLL}}(\theta_n)$$
  $$\theta_{\text{opt}} = \arg\min_\theta \mathcal{L}_{\text{NLL}}(\theta)$$
\end{frame}
