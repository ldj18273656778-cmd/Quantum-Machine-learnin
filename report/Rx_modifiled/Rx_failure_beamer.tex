\section{Motivation}
% ============================================================

\begin{frame}{Original Problem: Insufficient Generative Ability}
  \small
  \begin{center}
    \includegraphics[width=0.96\textwidth,height=0.56\textheight,keepaspectratio]{report/pattarn/assets/mnist_test_99_connectivity_bit_0_with_label_9_diagonal_threshold_0.4.png}
  \end{center}

  \vspace{-0.3em}
  \begin{itemize}
    \item Original encoding: $x_j=0 \rightarrow \ket{+}$, $x_j=1 \rightarrow \ket{0}$; $\ket{0}$ is a CZ eigenstate and acts as a barrier.
    \item The available ISQNN connectivity is fragmented into local components, while the diagonal label target is global and sparse.
    \item The original circuit does not possess the ability to generate global patterns.
  \end{itemize}
\end{frame}

\begin{frame}{Proposed Fix: $R_x(\phi)$ Gates}
  \[
  R_x(\phi)\ket{0} = \cos\frac{\phi}{2}\ket{0} - i\sin\frac{\phi}{2}\ket{1} \equiv \ket{u}
  \]
  \begin{itemize}
    \item Insert $R_x(\phi)$ after encoding, before $R_z(\theta)$
    \item $\ket{u}$ has $\ket{1}$ component $\Rightarrow$ CZ can now entangle with neighbors $\Rightarrow$ barrier broken
  \end{itemize}

  \vspace{0.4em}
  \begin{alertblock}{What is guaranteed, and what is not}
    The circuit's expressivity is certainly improved: $x=1$ is no longer a strict CZ barrier.
    However, whether this improvement is significant enough for MNIST, and whether it introduces new estimation or inference problems, must be verified.
  \end{alertblock}
\end{frame}

% ============================================================
\section{Parameter Estimation with $R_x$}
% ============================================================

\begin{frame}{Estimation When All Neighbors Are $x=1$}
  \small
  Consider target qubit $j$ with $x_j=0$, connected to $d$ CZ neighbors, all with $x=1$ ($\ket{0}\xrightarrow{R_x}\ket{u}$).

  \vspace{0.3em}
  After encoding, $R_x(\pi/4)$, $R_z(\theta_j)$, and CZ, the reduced density matrix of qubit $j$ \textbf{in the $Z$ basis} $\{\ket{0},\ket{1}\}$ (before final $H$ gate):
  \[
  \rho_j = \frac12\begin{pmatrix}
    1 & (\cos\phi)^d\, e^{-i\theta_j} \\[4pt]
    (\cos\phi)^d\, e^{i\theta_j} & 1
  \end{pmatrix}
  \]
  Converting to the $X$ basis via $\langle\pm|\rho_j|\pm\rangle$:
  \[
  P(y_j=1) = \frac{1}{2}\big[1 - (\cos\phi)^{d}\cos\theta_j\big]
  \]

  \textbf{Inverting:}
  \[
  \boxed{\hat\theta_j = \arccos\!\left(\frac{1 - 2\bar y_j}{(\cos\phi)^{d}}\right)}
  \]
  \begin{itemize}
    \item $d = |N(j)|$: number of CZ neighbors (varies per qubit)
    \item $\phi=0$ recovers the original formula: $\hat\theta_j = \arccos(1-2\bar y_j)$
  \end{itemize}
\end{frame}

\begin{frame}{Estimation Formula: Interpretation}
  \small
  \[
  \hat\theta_j = \arccos\!\left(\frac{1 - 2\bar y_j}{(\cos\phi)^{d}}\right)
  \]
  \begin{itemize}
    \item The numerator $1-2\bar y_j \in [-1,1]$ is the ``signal'' from the training data
    \item The denominator $(\cos\phi)^{d}$ \textbf{amplifies} the signal, compensating for the $R_x$-induced suppression
    \item Larger $d$ $\Rightarrow$ smaller $(\cos\phi)^d$ $\Rightarrow$ larger amplification
    \item For $\phi=\pi/4$: $\cos\phi = 1/\sqrt{2}$, so $(\cos\phi)^d = 2^{-d/2}$
  \end{itemize}

  \vspace{0.3em}
  \textbf{This is a theoretically valid closed-form estimator} --- if the training data $(x,y)$ were generated by the $R_x$-ISQNN circuit itself.

  \vspace{0.3em}
  The question is: does this formula remain useful when $y$ comes from an \emph{external} data source (MNIST images)?
\end{frame}

% ============================================================
\section{The Inference Gap}
% ============================================================

\begin{frame}{From Estimation to Inference: The Gap}
  \small
  \textbf{The natural workflow:}
  \begin{enumerate}
    \item Learn $\theta$ from training data $(x,y)$ using the estimation formula
    \item Use the learned $\theta$ to \textbf{infer} $y$ for new test inputs $x$
    \item Compare inferred $y$ with test labels to verify correctness
  \end{enumerate}

  \vspace{0.3em}
  \textbf{The problem:}
  \begin{itemize}
    \item Step 2 requires a circuit that accepts any $x$ and produces $y$ using the learned $\theta$
    \item The only circuit whose distribution \textbf{matches} the estimation formula is the $R_x$-ISQNN ($n_1m$ qubits)
    \item The $m$-qubit $R_x$-DQNN produces a \textbf{different distribution} (Reason 1)
  \end{itemize}

  \vspace{0.3em}
  \textbf{Central question:} Is there \emph{any} efficient DQNN structure that can correctly infer using parameters learned from the $R_x$-ISQNN?
\end{frame}

% ============================================================
\section{Why the Answer is No}
% ============================================================

\begin{frame}{Overview: Two Reasons}
  \begin{enumerate}
    \item \textbf{$R_x$ breaks the $R_z$-$CZ$ commutation} --- no classical rule can replicate ISQNN states
    \item \textbf{No low-qubit DQNN can replace ISQNN with $R_x$} --- Catch-22
  \end{enumerate}
\end{frame}

% ---- Reason 1 ----
\begin{frame}{Reason 1: Original Equivalence (Without $R_x$)}
  \small
  \textbf{Algebraic foundation:} $R_z$ and $CZ$ \textbf{commute} --- both diagonal in the $Z$ basis.
  \[ [R_z\otimes I,\; CZ] = 0,\qquad [I\otimes R_z,\; CZ] = 0 \]

  \begin{itemize}
    \item $x=1$ ($\ket{0}$): $R_z(\theta)\ket{0}=e^{-i\theta/2}\ket{0}$ (global phase). CZ eigenstate. \textbf{Stays $\ket{0}$.}
    \item $x=0$ ($\ket{+}$): $R_z(\theta)\ket{+}=\cos\frac{\theta}{2}\ket{+}-i\sin\frac{\theta}{2}\ket{-}$. CZ entangles, but 50/50 in $X$.
  \end{itemize}

  \vspace{0.3em}
  \textbf{Consequence:} Product state structure $\Rightarrow$ each qubit's evolution is independently trackable. DQNN can replicate ISQNN's post-measurement state using classical prescriptions ($H$+$X$ for $x=0$, measure+reset for $x=1$).
\end{frame}

\begin{frame}{Reason 1 (cont.): $R_x$ Breaks the Commutation Structure}
  \small
  \textbf{$R_x$ does NOT commute with $R_z$ or $CZ$:}
  \[ R_x(\phi) = \cos\frac{\phi}{2}\,I - i\sin\frac{\phi}{2}\,X \quad\text{(non-diagonal in $Z$ basis)} \]

  \begin{itemize}
    \item $x=1$: $\ket{0}\xrightarrow{R_x}\ket{u}$ --- \textbf{no longer a CZ eigenstate}
    \item After $R_z(\theta)\ket{u}$ and $CZ$, the qubit is in a \textbf{complex entangled state}
    \item When inter-slice $CZ$ acts, the qubit is \textbf{not} simply $\ket{0}$ or $\ket{+}$
    \item The evolution \textbf{cannot be factorized} into per-qubit tracks
  \end{itemize}

  \vspace{0.3em}
  \textbf{For DQNN:} Classical rules ($H+X$, reset, 50/50) assume the qubit is in a simple known state. With $R_x$, the actual state is complex and unknown to a classical prescription.
\end{frame}

\begin{frame}{Reason 1 (cont.): DQNN's Irreconcilable Dilemma}
  \small
  In ISQNN with $R_x$, multiple $\ket{u}$ qubits connected by CZ form an \textbf{entanglement cluster} spanning slices. DQNN faces two options --- both fail:

  \vspace{0.3em}
  \begin{center}
  \begin{tabular}{p{0.45\textwidth}|p{0.45\textwidth}}
    \toprule
    \textbf{Reset to $\ket{0}$} (current DQNN) & \textbf{Don't reset} \\
    \midrule
    After measuring $x=1$, reset removes $\ket{u}$ before subsequent $CZ$. & Keep qubit ``alive'' across blocks, accumulate all operations. \\
    \addlinespace
    \textbf{Result:} Inter-slice $\ket{u}$-$\ket{u}$ entanglement \textbf{destroyed}. DQNN sees isolated qubits. &
    \textbf{Result:} Output $y$ depends on accumulated $R_z$+$CZ$ history from $k-1$ blocks, not just $\theta_j$. \\
    \bottomrule
  \end{tabular}
  \end{center}

  \vspace{0.3em}
  \textbf{Dilemma:} Reset = entanglement gone. No reset = history accumulated. Neither reproduces ISQNN.
\end{frame}

% ---- Reason 2 ----
\begin{frame}{Reason 2: No Low-Qubit-Count DQNN Exists for $R_x$}
  \small
  \textbf{Without $R_x$:} $R_z$ and $CZ$ commute $\Rightarrow$ product-state structure $\Rightarrow$ a classical rule based DQNN ($m$ qubits) can reproduce the ISQNN ($n_1m$ qubits) distribution exactly.

  \textbf{With $R_x$:} Commutation broken $\Rightarrow$ qubit evolution is non-factorizable. The $\ket{u}$-$\ket{u}$ entanglement clusters \textbf{cannot} be decomposed into a sequential measurement + reset circuit.

  \vspace{0.3em}
  \begin{center}
  \begin{tabular}{lcc}
    \toprule
    & Without $R_x$ & With $R_x$ \\
    \midrule
    Algebraic structure & $[R_z,CZ]=0$ & $[R_x,CZ]\neq 0$, $[R_x,R_z]\neq 0$ \\
    Qubit tracking & 1 qubit at a time & Must track all simultaneously \\
    Low-qubit DQNN exists? & \textbf{Yes} (proven) & \textbf{No} (impossible) \\
    \bottomrule
  \end{tabular}
  \end{center}

  \vspace{0.3em}
  \textbf{The only correct circuit is ISQNN:} $n_1\times m = 100$ qubits, $2^{100}$ state vector $\Rightarrow$ \textbf{intractable}.
\end{frame}

\begin{frame}{Reason 2 (cont.): The Catch-22}
  \begin{itemize}
    \item MNIST requires $n_1=10$, $m=10$ $\Rightarrow$ 100 qubits
    \item ISQNN with $R_x$: $2^{100}$ state vector --- computationally \textbf{impossible}
    \item Any DQNN with $m=10$ qubits ($2^{10}=1024$ dim): \textbf{provably wrong} with $R_x$
  \end{itemize}

  \vspace{0.5em}
  \begin{center}
  \begin{tabular}{lcc}
    \toprule
    & ISQNN ($R_x$) & DQNN ($m=10$, $R_x$) \\
    \midrule
    State size & $2^{100}$ & $2^{10}=1024$ \\
    Correct distribution? & Yes & No (Reason 1) \\
    Computationally feasible? & \textbf{No} & Yes \\
    \bottomrule
  \end{tabular}
  \end{center}

  \vspace{0.5em}
  \textbf{Catch-22:} Correct circuit = intractable. Tractable circuit = incorrect. No middle ground exists.
\end{frame}

% ============================================================
\section{Conclusion}
% ============================================================

\begin{frame}{Summary}
  \begin{enumerate}
    \item $R_x$ breaks $R_z$-$CZ$ commutation $\Rightarrow$ \textbf{no low-qubit DQNN} can replicate $R_x$-ISQNN
    \item ISQNN intractable ($2^{100}$); DQNN wrong with $R_x$ --- \textbf{Catch-22}
  \end{enumerate}
\end{frame}

\begin{frame}{The Core Lesson}
  \centering
  \Large
  \textbf{Restoring connectivity is not enough.}

  \vspace{1em}
  \normalsize
  The $R_x$-$R_z$-$CZ$ algebraic incompatibility breaks the product-state structure that made the original DQNN--ISQNN equivalence possible. Without that structure, no efficient classical-quantum hybrid DQNN can replace the full $R_x$-ISQNN --- and the full ISQNN is computationally out of reach for $100$ qubits.
\end{frame}
